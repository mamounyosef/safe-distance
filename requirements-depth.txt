#   pip install --no-deps timm==1.0.30
#   pip install --no-deps "git+https://github.com/lpiccinelli-eth/UniDepth"
#   pip install --no-deps "git+https://github.com/ByteDance-Seed/depth-anything-3"
#
# Models are downloaded on first use to C:\safe-distance-models (override with
# the SAFE_DISTANCE_MODELS environment variable).
